cmake_minimum_required(VERSION 3.5)
project(ground_pp)
set(CMAKE_CXX_STANDARD 14)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(ground_pp_node node.cpp)
target_include_directories(ground_pp_node PRIVATE ${EIGEN3_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR})
ament_target_dependencies(ground_pp_node rclcpp nav_msgs sensor_msgs std_msgs ackermann_msgs)
install(TARGETS ground_pp_node DESTINATION lib/${PROJECT_NAME})
add_executable(health_interlock_node health_node.cpp)
ament_target_dependencies(health_interlock_node rclcpp std_msgs)
target_link_libraries(health_interlock_node jsoncpp)
add_executable(health_core_probe health_probe.cpp)
target_link_libraries(health_core_probe jsoncpp)
install(TARGETS health_interlock_node health_core_probe DESTINATION lib/${PROJECT_NAME})
add_executable(test_health_startup test_health_startup.cpp)
target_compile_options(test_health_startup PRIVATE -UNDEBUG)
target_link_libraries(test_health_startup jsoncpp)
install(TARGETS test_health_startup DESTINATION lib/${PROJECT_NAME})
ament_package()
