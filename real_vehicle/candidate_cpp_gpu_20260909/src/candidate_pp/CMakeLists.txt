cmake_minimum_required(VERSION 3.5)
project(candidate_pp)
set(CMAKE_CXX_STANDARD 14)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(candidate_pp_node node.cpp)
target_include_directories(candidate_pp_node PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(candidate_pp_node rclcpp nav_msgs sensor_msgs std_msgs ackermann_msgs)
add_executable(core_probe core_probe.cpp)
add_executable(core_tests core_tests.cpp)
install(TARGETS candidate_pp_node core_probe core_tests DESTINATION lib/${PROJECT_NAME})
ament_package()
