cmake_minimum_required(VERSION 3.5)
project(candidate_zero_pp)
set(CMAKE_CXX_STANDARD 14)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(candidate_zero_pp_node node.cpp)
target_include_directories(candidate_zero_pp_node PRIVATE ${EIGEN3_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/candidate_pp)
ament_target_dependencies(candidate_zero_pp_node rclcpp nav_msgs sensor_msgs std_msgs ackermann_msgs)
install(TARGETS candidate_zero_pp_node DESTINATION lib/${PROJECT_NAME})
ament_package()
