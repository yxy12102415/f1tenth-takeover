cmake_minimum_required(VERSION 3.5)
project(cpp_frenet)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
get_filename_component(JOINT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(JOINT_DEPS "${JOINT_ROOT}/../../reference_sources/joint_mpc_20260926/acados_install" ABSOLUTE)
include_directories(${JOINT_ROOT}/generated_joint ${JOINT_DEPS}/include ${JOINT_DEPS}/include/blasfeo/include ${JOINT_DEPS}/include/hpipm/include)
link_directories(${JOINT_ROOT}/lib)
set(CMAKE_BUILD_RPATH "${JOINT_ROOT}/lib")
set(CMAKE_INSTALL_RPATH "${JOINT_ROOT}/lib")
add_library(frenet_dynamic STATIC ../dynamic_mpc.cpp ../mpc_iteration.cpp)
target_include_directories(frenet_dynamic PUBLIC /usr/include/eigen3)
target_compile_options(frenet_dynamic PRIVATE -O3 -ffp-contract=off)
foreach(target cpp_frenet_planner cpp_frenet_controller)
 add_executable(${target} node.cpp)
 set_property(TARGET ${target} APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--disable-new-dtags")
 target_include_directories(${target} PRIVATE /usr/include/eigen3)
 target_compile_options(${target} PRIVATE -O3 -ffp-contract=off -Wall -Wextra)
 target_link_libraries(${target} frenet_dynamic jsoncpp pthread acados_ocp_solver_joint_vehicle acados hpipm blasfeo)
 ament_target_dependencies(${target} rclcpp nav_msgs sensor_msgs std_msgs ackermann_msgs visualization_msgs)
 install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
